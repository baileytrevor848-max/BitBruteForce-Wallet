import requests
import pandas as pd

wallets = open("wallets.txt").read().splitlines()

rows = []

for wallet in wallets:
    url = f"https://data-api.polymarket.com/positions?user={wallet}"
    positions = requests.get(url).json()

    for p in positions:
        rows.append({
            "wallet": wallet,
            "market": p.get("title"),
            "outcome": p.get("outcome"),
            "size": p.get("size"),
            "currentValue": p.get("currentValue"),
            "pnl": p.get("cashPnl")
        })

df = pd.DataFrame(rows)
df.to_csv("polymarket_wallet_positions.csv", index=False)

print(f"Exported {len(df)} positions")
